Synapse X UI Library

Classic Roblox UI Library :O

📦 Installation

local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/catminetry/Syllable-UI-Library/main/UI%20LIB.lua"))()

🪟 Window

local Window = Library:Init {
    name = "Synapse X - 1.0.2"
}

📑 Tabs

local Tab = Window:Tab({
    name = "Home",
    icon = "rbxassetid://10888331510"
})

🔘 Button

Tab:Button({
    name = "Click me",
    callback = function()
        print("Hello")
    end
})

⚠️ Warning

Tab:Warning({
    message = "This is a warning!"
})

🎚 Slider

Tab:Slider({
    title = "WalkSpeed",
    min = 16,
    max = 100,
    default = 16,
    callback = function(v)
        game.Players.LocalPlayer.Character.Humanoid.WalkSpeed = v
    end
})

✅ Toggle

Tab:Toggle({
    title = "Toggle",
    callback = function(v)
        print(v)
    end
})

📦 Dropdown

local Dropdown = Tab:Dropdown({
    title = "Dropdown",
    callback = function(v)
        print(v)
    end
})
Dropdown:Add("Option A", 1)
